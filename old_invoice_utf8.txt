const ProductCard = ({
  product,
  cart,
  onAdd,
  currencySymbol,
  selectedShopId,
}) => {
  const hasVariants =
    product.useVariants && product.variants && product.variants.length > 0;

  // Calculate variant stock for branch
  const getBranchStock = (itemStockArr, fallback) => {
    if (!selectedShopId || !itemStockArr) return fallback;
    const bInfo = itemStockArr.find(
      (b) => String(b.branchId?._id || b.branchId) === String(selectedShopId),
    );
    return bInfo ? bInfo.currentStock : 0;
  };

  const totalVariantStock = hasVariants
    ? product.variants.reduce(
        (sum, v) => sum + getBranchStock(v.branchStock, v.currentStock || 0),
        0,
      )
    : 0;

  const baseProductStock = getBranchStock(
    product.branchStock,
    product.currentStock || 0,
  );

  const totalCartQty = cart
    .filter((item) => item.itemType === "product" && item._id === product._id)
    .reduce((sum, item) => sum + item.qty, 0);

  const availableStock = hasVariants
    ? totalVariantStock - totalCartQty
    : baseProductStock - totalCartQty;

  // Tax calculations for non-variant display
  const taxPct = product.taxId?.percentage || product.taxPercent || 0;
  const isInclusive = product.isTaxInclusive || false;
  const basePrice = product.sellingPrice || 0;
  const taxAmt =
    taxPct > 0
      ? isInclusive
        ? Math.round(basePrice - basePrice / (1 + taxPct / 100))
        : Math.round(basePrice * (taxPct / 100))
      : 0;
  const effectivePrice = isInclusive ? basePrice : basePrice + taxAmt;

  // Price range for variant products
  let priceRange = null;
  if (hasVariants) {
    const prices = product.variants
      .map((v) => v.sellingPrice || 0)
      .filter((p) => p > 0);
    if (prices.length > 0) {
      const minPrice = Math.min(...prices);
      const maxPrice = Math.max(...prices);
      priceRange =
        minPrice === maxPrice
          ? `${currencySymbol}${Math.round(minPrice).toLocaleString()}`
          : `${currencySymbol}${Math.round(minPrice).toLocaleString()} - ${currencySymbol}${Math.round(maxPrice).toLocaleString()}`;
    }
  }

  const isOutOfStock = hasVariants
    ? totalVariantStock <= 0
    : availableStock <= 0;

  return (
    <div
      className={`bg-white dark:bg-dark-surface rounded-2xl overflow-hidden flex flex-col h-full group border ${isOutOfStock ? "opacity-60 cursor-not-allowed border-gray-100 dark:border-dark-border" : "cursor-pointer border-gray-100 dark:border-dark-border"}`}
      onClick={() => {
        if (!isOutOfStock) onAdd(product, product.sellingPrice);
      }}
    >
      <div className="h-32 bg-gray-50 dark:bg-dark-card relative overflow-hidden flex items-center justify-center">
        {product.image &&
        (product.image.url || typeof product.image === "string") ? (
          <img
            src={product.image.url || product.image}
            alt={product.name}
            className="w-full h-full object-cover"
          />
        ) : (
          <img
            src="https://png.pngtree.com/png-vector/20220119/ourmid/pngtree-crossed-camera-icon-image-is-not-available-message-restriction-forbidden-vector-png-image_44027682.jpg"
            alt="No Image"
            className="w-20 h-20 object-contain opacity-20"
          />
        )}
        {isOutOfStock && (
          <div className="absolute inset-0 bg-white/60 backdrop-blur-[1px] flex items-center justify-center">
            <span className="bg-red-500 text-white text-[10px] font-black uppercase tracking-widest px-3 py-1.5 rounded-full shadow-lg">
              Out of Stock
            </span>
          </div>
        )}
        {hasVariants && !isOutOfStock && (
          <div className="absolute top-2 right-2">
            <span className="bg-brand-primary text-white text-[8px] font-black uppercase tracking-widest px-2 py-1 rounded-full shadow-lg">
              {product.variants.length} Variants
            </span>
          </div>
        )}
      </div>
      <div className="p-3 flex-1 flex flex-col">
        <h4 className="font-bold text-gray-800 dark:text-gray-200 text-[13px] mb-1 line-clamp-2 leading-snug">
          {product.name}
        </h4>
        {taxPct > 0 && !hasVariants && (
          <span className="text-[8px] font-bold px-1.5 py-0.5 rounded bg-emerald-50 text-emerald-600 border border-emerald-100 w-fit mb-1">
            {product.taxId?.name || "Tax"} {taxPct}%
            {isInclusive ? " incl." : ""}
          </span>
        )}
        <div className="mt-auto flex items-center justify-between">
          <span
            className={`text-[11px] font-black uppercase tracking-tight px-2 py-0.5 rounded-full border ${availableStock <= 5 ? "bg-red-50 text-red-600 border-red-100" : "bg-emerald-50 text-emerald-600 border-emerald-100"}`}
          >
            {availableStock}
          </span>
          <div className="font-black text-sm text-brand-primary notranslate">
            {hasVariants ? (
              <span className="text-[11px]">
                {priceRange ||
                  `${currencySymbol}${Math.round(basePrice).toLocaleString()}`}
              </span>
            ) : (
              <>
                {currencySymbol}
                {Math.round(effectivePrice).toLocaleString()}
                {!isInclusive && taxPct > 0 && (
                  <span className="text-[8px] text-gray-400 font-bold ml-0.5">
                    (+tax)
                  </span>
                )}
              </>
            )}
          </div>
        </div>
      </div>
    </div>
  );
};

const VariantSelectionModal = ({
  isOpen,
  onClose,
  item,
  activeDept,
  departments,
  onSelect,
  currencySymbol,
  selectedShopId,
  cart,
}) => {
  const [selectedVariants, setSelectedVariants] = useState([]);

  // Reset selection when modal closes
  const handleClose = () => {
    setSelectedVariants([]);
    onClose();
  };

  if (!item) return null;

  const getVariantPriceForDept = (vInfo) => {
    // For Products: direct price fields
    if (item.itemType === "product") {
      return vInfo.sellingPrice || 0;
    }

    // For Services: check department-specific pricing first, then variant-level price, then service-level price
    let price = vInfo.price || item.price || 0;

    if (vInfo.pricing && vInfo.pricing.length > 0) {
      const po = vInfo.pricing.find((p) => {
        const deptId = (p.department?._id || p.department)?.toString();
        const activeDeptId = departments
          ?.find((d) => d.name === activeDept)
          ?._id?.toString();
        return (
          deptId === activeDeptId ||
          (p.department?.name && p.department.name === activeDept)
        );
      });
      if (po && po.price != null) price = po.price;
      else if (vInfo.pricing[0].price != null) price = vInfo.pricing[0].price;
    }

    return price;
  };

  const toggleVariant = (v, price) => {
    const vId = v._id || v.variantId?._id || v.variantId;
    setSelectedVariants((prev) => {
      const exists = prev.find((sv) => sv._id === vId);
      if (exists) return prev.filter((sv) => sv._id !== vId);
      return [
        ...prev,
        {
          _id: vId,
          name: v.variantId?.name || v.name || v.variantName || "Variant",
          price,
          taxId: v.taxId,
          taxPercent: v.taxPercent || item.taxPercent || 0,
          taxPercentage: v.taxId?.percentage || item.taxId?.percentage || 0,
          isTaxInclusive:
            v.isTaxInclusive !== undefined
              ? v.isTaxInclusive
              : item.isTaxInclusive || false,
          currentStock:
            item.itemType === "product"
              ? (() => {
                  if (!selectedShopId || !v.branchStock)
                    return v.currentStock || 0;
                  const bInfo = v.branchStock.find(
                    (b) =>
                      String(b.branchId?._id || b.branchId) ===
                      String(selectedShopId),
                  );
                  return bInfo ? bInfo.currentStock : 0;
                })()
              : 0,
        },
      ];
    });
  };

  const handleConfirm = () => {
    if (selectedVariants.length === 0) return;
    selectedVariants.forEach((v) => onSelect(item, v.price, v));
    handleClose();
  };

  return (
    <Modal
      isOpen={isOpen}
      onClose={handleClose}
      title="Select Variants"
      maxWidth="max-w-md"
    >
      <div className="space-y-6">
        {/* Item Header */}
        <div className="flex items-center gap-4 p-4 bg-gray-50/80 dark:bg-dark-card/50 rounded-2xl border border-gray-100 dark:border-dark-border">
          <div className="w-12 h-12 rounded-xl bg-brand-primary/10 flex items-center justify-center text-brand-primary">
            {item.itemType === "service" ? (
              <Scissors size={24} />
            ) : (
              <Package size={24} />
            )}
          </div>
          <div className="flex-1">
            <h4 className="font-bold text-[16px] text-gray-800 dark:text-gray-200">
              {item.name}
            </h4>
            <div className="flex items-center gap-2 mt-0.5">
              <span className="text-[10px] text-gray-400 font-bold uppercase tracking-widest">
                Choose Options
              </span>
              <div className="h-1 w-1 rounded-full bg-gray-300" />
              <span className="text-[10px] text-brand-primary font-bold">
                {activeDept ? `Pricing for ${activeDept}` : "Base Pricing"}
              </span>
            </div>
          </div>
        </div>

        {/* Variants List (Horizontal Row Wise) */}
        <div className="flex flex-wrap gap-3 max-h-[50vh] overflow-y-auto pr-1 custom-scrollbar">
          {item.variants?.map((v) => {
            const price = getVariantPriceForDept(v);
            const vId = v._id || v.variantId?._id || v.variantId;
            const isSelected = selectedVariants.some((sv) => sv._id === vId);
            const stockVal =
              item.itemType === "product"
                ? (() => {
                    if (!selectedShopId || !v.branchStock)
                      return v.currentStock || 0;
                    const bInfo = v.branchStock.find(
                      (b) =>
                        String(b.branchId?._id || b.branchId) ===
                        String(selectedShopId),
                    );
                    return bInfo ? bInfo.currentStock : 0;
                  })()
                : null;

            const cartQty = item.itemType === "product"
              ? cart.find((c) => c.itemType === "product" && (c.variantId === vId || c._id === vId))?.qty || 0
              : 0;

            const remainingStock = stockVal !== null ? stockVal - cartQty : null;

            return (
              <button
                key={vId}
                disabled={remainingStock !== null && remainingStock <= 0}
                onClick={() => toggleVariant(v, price)}
                className={`flex-1 min-w-[140px] flex flex-col items-start p-4 rounded-xl border group relative ${
                  remainingStock !== null && remainingStock <= 0
                    ? "opacity-50 grayscale cursor-not-allowed bg-gray-50 dark:bg-dark-card border-gray-200"
                    : isSelected
                      ? "bg-brand-primary/10 border-brand-primary shadow-lg shadow-brand-primary/5 ring-1 ring-brand-primary/20"
                      : "bg-white dark:bg-dark-surface border-gray-100 dark:border-dark-border"
                }`}
              >
                {(() => {
                  const matchingPricing = v.pricing?.find((p) => {
                    const dName =
                      p.department?.name ||
                      departments.find(
                        (d) => d._id === (p.department?._id || p.department),
                      )?.name;
                    return dName === activeDept;
                  });
                  const hasSpecificDeptPrice =
                    matchingPricing &&
                    matchingPricing.price != null &&
                    matchingPricing.price !== "";

                  return hasSpecificDeptPrice ? (
                    <div className="absolute -top-2 -right-1 bg-brand-primary text-white text-[7px] font-black px-1.5 py-0.5 rounded-full shadow-sm flex items-center gap-1">
                      <Check size={8} /> {activeDept}
                    </div>
                  ) : (
                    <div className="absolute -top-2 -right-1 bg-gray-200 text-gray-500 text-[6px] font-bold px-1.5 py-0.5 rounded-full shadow-sm">
                      Base Price
                    </div>
                  );
                })()}
                <div className="flex items-center justify-between w-full mb-1">
                  <span
                    className={`text-[13px] font-black uppercase tracking-tight ${isSelected ? "text-brand-primary" : "text-gray-700 dark:text-gray-300 group-hover:text-brand-primary"}`}
                  >
                    {v.variantId?.name || v.name || v.variantName || "Variant"}
                  </span>
                  {isSelected && (
                    <div className="bg-brand-primary text-white p-0.5 rounded-full ring-4 ring-brand-primary/10">
                      <Check size={10} strokeWidth={4} />
                    </div>
                  )}
                </div>

                <div className="flex items-baseline gap-1">
                  <span
                    className={`text-[16px] font-black notranslate ${isSelected ? "text-brand-primary" : "text-gray-900 dark:text-gray-100"}`}
                  >
                    {currencySymbol}
                    {price.toLocaleString()}
                  </span>
                  {(v.taxId?.percentage > 0 ||
                    v.taxPercent > 0 ||
                    item.taxId?.percentage > 0 ||
                    item.taxPercentage > 0) && (
                    <div className="flex flex-col items-end">
                      <span className="text-[10px] font-bold text-gray-500 uppercase tracking-widest leading-none mb-0.5">
                        Tax Amount
                      </span>
                      <span className="text-[10px] font-black text-emerald-600 notranslate">
                        {currencySymbol}
                        {Math.round(
                          price *
                            ((v.taxId?.percentage ||
                              v.taxPercent ||
                              item.taxId?.percentage ||
                              item.taxPercentage ||
                              0) /
                              100) *
                            (v.isTaxInclusive || item.isTaxInclusive ? 0 : 1),
                        )}
                        <span className="text-[7px] ml-0.5">
                          (
                          {v.taxId?.percentage ||
                            v.taxPercent ||
                            item.taxId?.percentage ||
                            item.taxPercentage ||
                            0}
                          %)
                        </span>
                      </span>
                    </div>
                  )}
                </div>

                {remainingStock !== null && (
                  <div
                    className={`mt-2 text-[10px] font-black uppercase tracking-tight px-2.5 py-1 rounded-full border shadow-sm ${remainingStock <= 5 ? "bg-red-50 text-red-600 border-red-200" : "bg-emerald-50 text-emerald-600 border-emerald-200"}`}
                  >
                    {remainingStock}
                  </div>
                )}
              </button>
            );
          })}
        </div>

        {/* Action Buttons */}
        <div className="grid grid-cols-2 gap-3 pt-2">
          <button
            onClick={handleClose}
            className="py-4 rounded-2xl bg-gray-50 dark:bg-dark-card text-gray-400 font-bold text-[12px] uppercase tracking-[0.2em] hover:bg-gray-100 dark:hover:bg-dark-border transition-all"
          >
            Cancel
          </button>
          <button
            onClick={handleConfirm}
            disabled={selectedVariants.length === 0}
            className={`py-4 rounded-2xl font-bold text-[12px] uppercase tracking-[0.2em] transition-all ${
              selectedVariants.length > 0
                ? "bg-brand-primary text-white shadow-lg shadow-brand-primary/25 hover:opacity-90 active:scale-95"
                : "bg-gray-100 dark:bg-dark-border text-gray-300 cursor-not-allowed"
            }`}
          >
            Add ({selectedVariants.length})
          </button>
        </div>
      </div>
    </Modal>
  );
};

        const res = await api.get("/invoices/metadata");
        if (res.data?.success) {
          data = res.data.data;
          metadataCache = data;
          applyMetadata(data);
        }

        if (!data) return;

        const fetchedServices = data.services || [];
        const fetchedDepts = data.departments || [];

        // Handle Redirection State (Pre-fill from Booking)
        if (location.state?.booking) {
          const booking = location.state.booking;
          if (booking.user) setSelectedCustomer(booking.user);
          if (booking.departmentId) {
            const deptName =
              booking.departmentId.name ||
              fetchedDepts.find(
                (d) => d._id?.toString() === booking.departmentId?.toString(),
              )?.name;
            if (deptName) setActiveDept(deptName);
          }
          if (booking.services?.length > 0) {
            const bookingCart = [
              ...booking.services.map((s) => {
                const sId = s.serviceId?._id || s.serviceId || s._id || s;
                const fullService = fetchedServices.find(
                  (fs) => fs._id === sId.toString(),
                );
                const variantId = s.variantId?._id || s.variantId;
                let deptName = "";
                let deptIdStr = "";

                if (booking.departmentId) {
                  deptName =
                    booking.departmentId.name ||
                    fetchedDepts.find(
                      (d) =>
                        d._id?.toString() === booking.departmentId?.toString(),
                    )?.name ||
                    "";
                  deptIdStr = (
                    booking.departmentId._id || booking.departmentId
                  ).toString();
                }

                let variantName = "";
                let price = s.price || fullService?.price || 0;

                if (fullService && variantId && variantId !== "base") {
                  const vidStr = variantId.toString();
                  const variantObj = fullService.variants?.find((v) => {
                    const vvid = (
                      v.variantId?._id ||
                      v.variantId?.id ||
                      v.variantId ||
                      v._id
                    )?.toString();
                    return vvid === vidStr;
                  });

                  if (variantObj) {
                    variantName = variantObj.variantId?.name || "";
                    const vPricing = variantObj.pricing?.find((p) => {
                      const pDeptId = (
                        p.department?._id ||
                        p.department ||
                        p.department?.id
                      )?.toString();
                      return pDeptId === deptIdStr;
                    });
                    price =
                      vPricing?.price ||
                      variantObj.price ||
                      variantObj.pricing?.[0]?.price ||
                      price;
                  }
                } else if (fullService?.pricing?.length > 0) {
                  let po = fullService.pricing.find((p) => {
                    const pdId = (
                      p.department?._id || p.department
                    )?.toString();
                    return (
                      pdId === deptIdStr ||
                      (p.department?.name && p.department.name === deptName)
                    );
                  });
                  if (po && po.price != null) price = po.price;
                }

                return {
                  ...(fullService || s),
                  _id: sId,
                  price,
                  qty: 1,
                  variantId: variantId || "base",
                  variantName,
                  variantKey: `svc-${sId}-${variantId || "base"}-${price}-${deptName}`,
                  deptName,
                  fromBookingId: booking._id,
                  itemType: "service",
                  taxName: fullService?.taxId?.name || "",
                  taxPercentage: fullService?.taxId?.percentage || 0,
                  isTaxInclusive: fullService?.isTaxInclusive || false,
                };
              }),
              ...(booking.products || []).map((p) => ({
                ...p.product,
                _id: p.product?._id || p.product,
                price: p.price || p.product?.sellingPrice || 0,
                qty: p.quantity || 1,
                variantKey: `prod-${p.product?._id || p.product}-base-${p.price || p.product?.sellingPrice || 0}`,
                itemType: "product",
                fromBookingId: booking._id,
              })),
            ];
            setCart(bookingCart);
          }
          if (booking.branchId)
            setSelectedShopId(
              (booking.branchId._id || booking.branchId).toString(),
            );
          if (booking.staff)
            setSelectedStaffId(booking.staff._id || booking.staff);
        }

        // Handle Edit Mode Redirection (From Invoice Report)
        const queryParams = new URLSearchParams(location.search);
        const editId = queryParams.get("edit");
        if (editId) {
          setIsEditing(true);
          setEditInvoiceId(editId);
          try {
            const invRes = await api.get(`/invoices/${editId}`);
            if (invRes.data?.success) {
              const invoice = invRes.data.data;
              if (invoice.user) setSelectedCustomer(invoice.user);
              if (invoice.shopId)
                setSelectedShopId(invoice.shopId._id || invoice.shopId);
              if (invoice.branchId)
                setSelectedShopId(invoice.branchId._id || invoice.branchId);
              if (invoice.staff)
                setSelectedStaffId(invoice.staff._id || invoice.staff);
              if (invoice.discount) setDiscount(invoice.discount.toString());
              if (invoice.payments && invoice.payments.length > 0) {
                setPayments(invoice.payments);
              } else if (invoice.paymentMethod) {
                setPayments([
                  {
                    method: invoice.paymentMethod,
                    amount: invoice.paidAmount || invoice.amount,
                  },
                ]);
                setPaymentMethod(invoice.paymentMethod);
              }

              let loadedCartMap = new Map();

              if (invoice.appointment && invoice.appointment.services) {
                let deptName = "";
                if (invoice.appointment.departmentId) {
                  const foundDept = fetchedDepts.find(
                    (d) =>
                      d._id?.toString() ===
                      invoice.appointment.departmentId?.toString(),
                  );
                  if (foundDept) deptName = foundDept.name;
                }
                if (!deptName && fetchedDepts.length > 0) {
                  deptName = fetchedDepts[0].name;
                }
                if (deptName) setActiveDept(deptName);

                invoice.appointment.services.forEach((s) => {
                  const sId =
                    typeof s === "object"
                      ? s.serviceId?._id || s.serviceId || s._id
                      : s;
                  const variantId =
                    typeof s === "object"
                      ? s.variantId?._id || s.variantId
                      : null;
                  const fullService = fetchedServices.find(
                    (fs) => fs._id === sId,
                  );

                  let price =
                    typeof s === "object" && s.price
                      ? s.price
                      : fullService?.price || 0;

                  if (deptName && fullService?.pricing?.length > 0) {
                    const po = fullService.pricing.find(
                      (p) => p.department?.name === deptName,
                    );
                    if (po && po.price != null) price = po.price;
                    else if (fullService.price != null)
                      price = fullService.price;
                  }

                  const variantKey = `svc-${sId}-${price}-${deptName}`;

                  if (loadedCartMap.has(variantKey)) {
                    loadedCartMap.get(variantKey).qty += 1;
                  } else {
                    loadedCartMap.set(variantKey, {
                      ...(fullService || s),
                      _id: sId,
                      price: price,
                      qty: 1,
                      variantKey: variantKey,
                      variantId: variantId || "base",
                      variantName:
                        s.variantName ||
                        fullService?.variants?.find(
                          (v) =>
                            (v._id || v.variantId?._id || v.variantId) ===
                            variantId,
                        )?.variantId?.name ||
                        "",
                      deptName: deptName,
                      itemType: "service",
                      taxName: fullService?.taxId?.name || "",
                      taxPercentage: fullService?.taxId?.percentage || 0,
                      taxRate: fullService?.taxId?.percentage || 0,
                      isTaxInclusive: fullService?.isTaxInclusive || false,
                    });
                  }
                });
              }

              if (invoice.items && invoice.items.length > 0) {
                invoice.items.forEach((pi) => {
                  const pId = pi.product?._id || pi.product;
                  const fullProduct =
                    products.find((p) => p._id === pId) || pi.product || {};

                  const variantKey = `prod-${pId}`;

                  if (loadedCartMap.has(variantKey)) {
                    loadedCartMap.get(variantKey).qty += pi.quantity;
                  } else {
                    loadedCartMap.set(variantKey, {
                      ...fullProduct,
                      _id: pId,
                      name:
                        pi.productName || fullProduct.name || "Unknown Product",
                      price: pi.price,
                      qty: pi.quantity,
                      variantKey: variantKey,
                      itemType: "product",
                      taxPercent: pi.taxPercent || 0,
                    });
                  }
                });
              }
              setCart(Array.from(loadedCartMap.values()));
            }
          } catch (err) {
            console.error("Failed to load invoice for editing", err);
            showSnack("Failed to load invoice details", "error");
          }
        }
      } catch (err) {
        console.error("Failed to load initial data", err);
      } finally {
        setLoading(false);
      }
    };
    fetchData();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // --- Filtered Data ---
  const filteredStaff = useMemo(() => {
    if (!selectedShopId) return staff;
    return staff.filter((s) => {
      const bid = s.branchId?._id || s.branchId || s.shop?._id || s.shop;
      return String(bid) === String(selectedShopId);
    });
  }, [staff, selectedShopId]);

  const filteredServices = useMemo(() => {
    return services.filter((service) => {
      if (!service.isActive) return false;

      // Restrict services by branch
      const branchToFilter = isAdmin
        ? selectedShopId
        : user?.branchId?._id || user?.branchId;
      if (branchToFilter) {
        const hasBranchAccess =
          !service.branches ||
          service.branches.length === 0 ||
          service.branches.some(
            (b) => (b._id || b).toString() === String(branchToFilter),
          );
        if (!hasBranchAccess) return false;
      }

      if (
        searchQuery &&
        !service.name.toLowerCase().includes(searchQuery.toLowerCase())
      )
        return false;
      if (showOnlyFavourites && !service.isFavourite) return false;
      if (activeDept) {
        const hasTopLevelPricing = service.pricing?.some(
          (p) =>
            p.department?.name === activeDept &&
            p.price != null &&
            p.price !== "",
        );
        const hasVariantPricing =
          service.useVariants &&
          service.variants?.some((v) =>
            v.pricing?.some((p) => {
              const dName =
                p.department?.name ||
                departments.find(
                  (d) => d._id === (p.department?._id || p.department),
                )?.name;
              return dName === activeDept && p.price != null && p.price !== "";
            }),
          );

        if (!hasTopLevelPricing && !hasVariantPricing) return false;
      }
      if (activeSvcCat !== "All") {
        if (!service.categories?.some((c) => c.name === activeSvcCat))
          return false;
      }
      return true;
    });
  }, [
    services,
    activeDept,
    activeSvcCat,
    searchQuery,
    showOnlyFavourites,
    isAdmin,
    selectedShopId,
    departments,
    user,
  ]);

  const filteredProducts = useMemo(() => {
    return products.filter((product) => {
      if (product.isActive === false) return false;

      // Restrict products by branch
      const branchToFilter = isAdmin
        ? selectedShopId
        : user?.branchId?._id || user?.branchId;
      if (branchToFilter) {
        const hasBranchAccess =
          !product.branches ||
          product.branches.length === 0 ||
          product.branches.some(
            (b) => (b._id || b).toString() === String(branchToFilter),
          );
        if (!hasBranchAccess) return false;
      }

      if (
        searchQuery &&
        !product.name.toLowerCase().includes(searchQuery.toLowerCase()) &&
        !product.sku?.toLowerCase().includes(searchQuery.toLowerCase())
      )
        return false;
      if (activeProdCat !== "All" && product.category !== activeProdCat)
        return false;
      if (product.isSale === false) return false;
      return true;
    });
  }, [products, activeProdCat, searchQuery, isAdmin, selectedShopId, user]);

  const getServicePrice = (service) => {
    if (!service) return 0;

    if (activeDept) {
      const topLevelPo = service.pricing?.find(
        (p) => p.department?.name === activeDept,
      );
      if (topLevelPo && topLevelPo.price != null && topLevelPo.price !== "") {
        return topLevelPo.price;
      } else if (service.useVariants && service.variants?.length > 0) {
        // Find min price among variants for this department
        const variantPrices = service.variants
          .map((v) => {
            const vp = v.pricing?.find((p) => {
              const dName =
                p.department?.name ||
                departments.find(
                  (d) => d._id === (p.department?._id || p.department),
                )?.name;
              return dName === activeDept;
            });
            return vp && vp.price != null && vp.price !== "" ? vp.price : 0;
          })
          .filter((p) => p > 0);
        if (variantPrices.length > 0) return Math.min(...variantPrices);
      }
      return 0; // No price for this department
    }

    // Default price resolution when no department is active
    let price = service.price || 0;
    if (service.useVariants && service.variants?.length > 0) {
      const variantPrices = service.variants
        ?.map((v) => v.price || v.pricing?.[0]?.price || 0)
        .filter((p) => p > 0);
      if (variantPrices?.length > 0) {
        price = Math.min(...variantPrices);
      }
    }

    return price;
  };

  const filteredCustomers = useMemo(() => {
    if (!customerSearch) return [];
    const lower = customerSearch.toLowerCase();
    return customers
      .filter(
        (c) =>
          c.name.toLowerCase().includes(lower) ||
          c.phone.includes(lower) ||
          (c.email && c.email.toLowerCase().includes(lower)),
      )
      .slice(0, 5);
  }, [customers, customerSearch]);

  // --- Cart Calculations (handles both service and product tax models) ---
  const cartCalc = useMemo(() => {
    let subtotal = 0,
      totalTax = 0,
      hasInclusiveTax = false;
    cart.forEach((item) => {
      const lineTotal = item.price * item.qty;
      subtotal += lineTotal;

      const taxPct =
        item.itemType === "service"
          ? item.taxPercentage || 0
          : item.taxPercent || 0;

      if (taxPct > 0) {
        if (!item.isTaxInclusive) {
          totalTax += lineTotal * (taxPct / 100);
        } else {
          hasInclusiveTax = true;
        }
      }
    });
    return {
      subtotal: Math.round(subtotal),
      totalTax: Math.round(totalTax),
      hasInclusiveTax,
    };
  }, [cart]);

  const cartTotal = Math.round(cartCalc.subtotal + cartCalc.totalTax);
  const finalTotal = Math.max(
    0,
    Math.round(cartTotal - (Number(discount) || 0)),
  );

  // --- Handlers ---
  const addServiceToCart = (service, price, variantData = null) => {
    if (service.useVariants && !variantData) {
      setItemForVariants({ ...service, itemType: "service" });
      setShowVariantModal(true);
      return;
    }

    setCart((prev) => {
      const variantId = variantData ? variantData._id : "base";
      const variantName = variantData ? variantData.name : "";
      const finalPrice = Number(price) || 0;
      const variantKey = `svc-${service._id}-${variantId}-${finalPrice}-${activeDept}`;

      if (prev.find((item) => item.variantKey === variantKey)) {
        showSnack(`${service.name} ${variantName} already in cart`, "info");
        return prev;
      }

      const variantTaxId = variantData?.taxId?._id || variantData?.taxId;
      const variantTax = variantTaxId
        ? taxes.find((t) => t._id === variantTaxId) || variantData.taxId
        : null;

      const matchingPricing = variantData
        ? variantData.pricing?.find(
            (p) =>
              (p.department?.name ||
                taxes.find((d) => d._id === (p.department?._id || p.department))
                  ?.name) === activeDept,
          )
        : service.pricing?.find((p) => p.department?.name === activeDept);

      const isDeptMatched = !!(
        matchingPricing &&
        matchingPricing.price != null &&
        matchingPricing.price !== ""
      );

      return [
        ...prev,
        {
          ...service,
          price: finalPrice,
          qty: 1,
          variantKey,
          variantId,
          variantName,
          deptName: activeDept,
          isDeptMatched,
          itemType: "service",
          taxName:
            variantTax?.name ||
            variantData?.taxName ||
            service.taxId?.name ||
            service.taxName ||
            "",
          taxPercentage:
            variantTax?.percentage ??
            variantData?.taxPercentage ??
            variantData?.taxPercent ??
            service.taxId?.percentage ??
            service.taxPercentage ??
            0,
          isTaxInclusive:
            variantData?.isTaxInclusive !== undefined
              ? variantData.isTaxInclusive
              : service.isTaxInclusive || false,
        },
      ];
    });
    setShowVariantModal(false);
  };

  const addProductToCart = (product, price, variantData = null) => {
    if (product.useVariants && !variantData) {
      setItemForVariants({ ...product, itemType: "product" });
      setShowVariantModal(true);
      return;
    }

    const finalPrice = product.useVariants
      ? Number(price) || 0
      : product.sellingPrice || Number(price) || 0;

    setCart((prev) => {
      const variantId = variantData
        ? variantData._id || variantData.variantId?._id || variantData.variantId
        : "base";
      const variantName = variantData
        ? variantData.name || variantData.variantId?.name || ""
        : "";
      const variantKey = `prod-${product._id}-${variantId}-${finalPrice}`;
      const existing = prev.find((item) => item.variantKey === variantKey);

      if (existing) {
        if (existing.qty + 1 > existing.currentStock) {
          showSnack(
            `Cannot add more than available stock (${existing.currentStock}).`,
            "warning",
          );
          return prev;
        }
        return prev.map((item) =>
          item.variantKey === variantKey
            ? { ...item, qty: item.qty + 1 }
            : item,
        );
      }
      return [
        ...prev,
        {
          ...product,
          price: finalPrice || 0,
          qty: 1,
          variantKey,
          variantId,
          variantName,
          itemType: "product",
          taxName:
            variantData?.taxId?.name ||
            (variantData?.taxId
              ? taxes.find((t) => t._id === variantData.taxId)?.name
              : null) ||
            product.taxId?.name ||
            "",
          taxPercent:
            variantData?.taxId?.percentage ||
            variantData?.taxPercentage ||
            variantData?.taxPercent ||
            product.taxId?.percentage ||
            product.taxPercent ||
            0,
          taxPercentage:
            variantData?.taxId?.percentage ||
            variantData?.taxPercentage ||
            variantData?.taxPercent ||
            product.taxId?.percentage ||
            product.taxPercent ||
            0,
          isTaxInclusive:
            variantData?.isTaxInclusive !== undefined
              ? variantData.isTaxInclusive
              : product.isTaxInclusive || false,
          currentStock: (() => {
            const dataSource = variantData || product;
            if (!selectedShopId || !dataSource.branchStock)
              return dataSource.currentStock || 0;
            const bInfo = dataSource.branchStock.find(
              (b) =>
                String(b.branchId?._id || b.branchId) ===
                String(selectedShopId),
            );
            return bInfo ? bInfo.currentStock : 0;
          })(),
        },
      ];
    });
    setShowVariantModal(false);
  };

  const updateQty = (variantKey, delta) => {
    setCart((prev) =>
      prev
        .map((item) => {
          if (item.variantKey === variantKey) {
            const newQty = item.qty + delta;
            if (item.itemType === "product" && newQty > item.currentStock) {
              showSnack(
                `Only ${item.currentStock} available in stock.`,
                "warning",
              );
              return item;
            }
            return newQty > 0 ? { ...item, qty: newQty } : null;
          }
          return item;
        })
        .filter(Boolean),
    );
  };

  const removeFromCart = (variantKey) =>
    setCart((prev) => prev.filter((item) => item.variantKey !== variantKey));

  const handleAddPaymentRow = () => {
    const remaining =
      finalTotal -
      payments.reduce((sum, p) => sum + (Number(p.amount) || 0), 0);
    setPayments([
      ...payments,
      {
        method: paymentMethods[0]?.name || "CASH",
        amount: remaining > 0 ? remaining : "",
      },
    ]);
  };

  const handleRemovePaymentRow = (index) =>
    setPayments(payments.filter((_, i) => i !== index));

  const handleUpdatePayment = (index, field, value) => {
    setPayments((prev) => {
      const np = [...prev];
      np[index] = { ...np[index], [field]: value };
      if (field === "amount" && np.length > 1) {
        const uv = Number(value) || 0;
        if (np.length === 2) {
          const oi = index === 0 ? 1 : 0;
          np[oi] = { ...np[oi], amount: Math.max(0, finalTotal - uv) || "" };
        } else {
          const ti = index === np.length - 1 ? np.length - 2 : np.length - 1;
          const to = np.reduce(
            (s, p, i) =>
              i === ti || i === index ? s : s + (Number(p.amount) || 0),
            0,
          );
          np[ti] = {
            ...np[ti],
            amount: Math.max(0, finalTotal - to - uv) || "",
          };
        }
      }
      return np;
    });
  };

  const remainingAmount = useMemo(() => {
    const paid = payments.reduce((sum, p) => sum + (Number(p.amount) || 0), 0);
    return finalTotal - paid;
  }, [payments, finalTotal]);

  useEffect(() => {
    if (finalTotal > 0) {
      setPayments((prev) => {
        if (prev.length === 0)
          return [
            { method: paymentMethods[0]?.name || "CASH", amount: finalTotal },
          ];
        if (prev.length === 1) return [{ ...prev[0], amount: finalTotal }];
        const ctp = prev.reduce((s, p) => s + (Number(p.amount) || 0), 0);
        if (ctp > finalTotal) {
          const np = [...prev];
          const li = np.length - 1;
          const tel = np
            .slice(0, -1)
            .reduce((s, p) => s + (Number(p.amount) || 0), 0);
          np[li].amount = Math.max(0, finalTotal - tel);
          return np;
        }
        return prev;
      });
    }
  }, [finalTotal, paymentMethods]);

  const handleCustomerSelect = (cust) => {
    setSelectedCustomer(cust);
    setCustomerSearch("");
    setShowCustomerResults(false);
  };

  const handleCreateCustomer = async (e) => {
    e.preventDefault();
    try {
      const formData = new FormData();
      formData.append("name", newCustData.name);
      formData.append("phone", newCustData.phone);
      if (newCustData.email) formData.append("email", newCustData.email);
      const res = await api.post("/admin/customers", formData, {
        headers: { "Content-Type": "multipart/form-data" },
      });
      if (res.data?.success) {
        const nc = res.data.data;
        setCustomers((prev) => [...prev, nc]);
        setSelectedCustomer(nc);
        setIsCustModalOpen(false);
        setNewCustData({ name: "", phone: "", email: "" });
        showSnack("Customer Created & Selected", "success");
      }
    } catch (err) {
      showSnack(
        err.response?.data?.message || "Failed to create customer",
        "error",
      );
    }
  };

  const handleBookAppointment = async () => {
    const serviceItems = cart.filter((i) => i.itemType === "service");
    if (serviceItems.length === 0)
      return showSnack(
        "Please add services to the cart for booking.",
        "warning",
      );

    try {
      setLoading(true);

      let serviceItemsPayload = [];
      const productItems = cart.filter((i) => i.itemType === "product");
      serviceItems.forEach((item) => {
        for (let i = 0; i < item.qty; i++) {
          serviceItemsPayload.push({
            serviceId: item._id,
            variantId: item.variantId !== "base" ? item.variantId : undefined,
          });
        }
      });

      if (!bookingData.time) {
        return showSnack("Please select a slot time", "warning");
      }

      // Send just the selected slot - backend will normalize for multi-service
      const slotTimes = [bookingData.time];

      const shop =
        selectedShopId || (branches.length === 1 ? branches[0]._id : null);

      const dominantDeptName = serviceItems[0]?.deptName || activeDept;
      const deptId = departments.find((d) => d.name === dominantDeptName)?._id;

      const payload = {
        shopId: shop,
        branchId: shop,
        departmentId: deptId,
        services: serviceItemsPayload,
        date: bookingData.date,
        slotTimes,
        customerId: selectedCustomer ? selectedCustomer._id : null,
        customerDetails: selectedCustomer
          ? {
              name: selectedCustomer.name,
              phone: selectedCustomer.phone,
              email: selectedCustomer.email,
            }
          : null,
        staffId: selectedStaffId === "admin" ? null : selectedStaffId,
        isDirectInvoice: false,
        items: productItems.map((p) => ({
          product: p._id,
          variantId: p.variantId !== "base" ? p.variantId : undefined,
          quantity: p.qty,
          price: p.price,
        })),
      };

      const res = await api.post("/appointments", payload);
      if (res.data?.success) {
        showSnack("Appointment Booked Successfully", "success");
        setIsBookingModalOpen(false);
        setCart([]); // Clear cart
      }
    } catch (err) {
      console.error("Booking error:", err);
      showSnack(
        err.response?.data?.message || "Failed to book appointment",
        "error",
      );
    } finally {
      setLoading(false);
    }
  };

  const handleCheckout = async () => {
    if (cart.length === 0) return showSnack("Cart is empty.", "warning");
    if (branches.length > 1 && !selectedShopId)
      return showSnack("Please select a branch.", "warning");

    const walletPayment = payments.find((p) => p.method === "WALLET");
    // Wallet Payment Logic - No longer blocking, just ensuring customer is selected
    if (walletPayment && Number(walletPayment.amount) > 0) {
      if (!selectedCustomer)
        return showSnack(
          "Customer must be selected for Wallet payment.",
          "warning",
        );
    }

    const shop =
      selectedShopId || (branches.length === 1 ? branches[0]._id : null);
    const serviceItems = cart.filter((i) => i.itemType === "service");
    const productItems = cart.filter((i) => i.itemType === "product");

    try {
      setLoading(true);

      let serviceItemsPayload = [];
      let slotTimes = [];
      let bookingId = null;

      if (serviceItems.length > 0) {
        serviceItems.forEach((item) => {
          for (let i = 0; i < item.qty; i++) {
            serviceItemsPayload.push({
              serviceId: item._id,
              variantId: item.variantId !== "base" ? item.variantId : undefined,
              price: item.price,
              variantName: item.variantName || null,
              taxName: item.taxName || "",
              taxPercentage: item.taxPercentage || 0,
              isTaxInclusive: item.isTaxInclusive || false,
            });
          }
        });
        const now = new Date();
        const timeString = `${now.getHours().toString().padStart(2, "0")}:${now.getMinutes().toString().padStart(2, "0")}`;
        slotTimes = serviceItemsPayload.map(() => timeString);

        if (!isEditing) {
          bookingId = serviceItems.find(
            (item) => item.fromBookingId,
          )?.fromBookingId;
        }
      }

      const items = productItems.map((item) => ({
        product: item._id,
        variantId: item.variantId !== "base" ? item.variantId : undefined,
        productName: item.name,
        variantName: item.variantName,
        quantity: item.qty,
        price: item.price,
        taxPercent: item.taxPercent || item.taxPercentage || 0,
        taxPercentage: item.taxPercentage || item.taxPercent || 0,
        taxAmount:
          item.price *
          item.qty *
          ((item.taxPercent || item.taxPercentage || 0) / 100),
        amount: item.price * item.qty,
        itemType: "product",
        taxName:
          item.taxName || `Tax ${item.taxPercent || item.taxPercentage || 0}%`,
        isTaxInclusive: item.isTaxInclusive || false,
      }));

      const dominantDeptName = serviceItems[0]?.deptName || activeDept;
      const deptId = departments.find((d) => d.name === dominantDeptName)?._id;

      const payload = {
        shopId: shop,
        branchId: shop,
        departmentId: deptId,
        services: serviceItemsPayload,
        items: items,
        date: moment().format("YYYY-MM-DD"),
        slotTimes,
        customerId: selectedCustomer ? selectedCustomer._id : null,
        customerDetails: selectedCustomer
          ? {
              name: selectedCustomer.name,
              phone: selectedCustomer.phone,
              email: selectedCustomer.email,
            }
          : {
              name: newCustData.name || "LOCAL CUSTOMER",
              phone: newCustData.phone || "0000000000",
              email: newCustData.email || "",
            },
        paymentMethod:
          payments.length > 1 ? "Split" : payments[0]?.method || paymentMethod,
        payments: payments,
        discount: Number(discount) || 0,
        finalAmount: finalTotal,
        taxAmount: cartCalc.totalTax,
        isDirectInvoice: true,
        staffId:
          isAdmin && selectedStaffId !== "admin" ? selectedStaffId : undefined,
        amount: cartCalc.subtotal,
      };

      let responseData = null;
      if (isEditing && editInvoiceId) {
        // Just calling a backend update for the invoice instead of creating new appointments or completing them
        const res = await api.put(`/invoices/${editInvoiceId}`, {
          ...payload,
          items: items,
          services: serviceItemsPayload,
        });
        responseData = res.data?.data;
      } else if (bookingId) {
        const res = await api.put(`/appointments/${bookingId}/complete`, {
          paymentMethod: payload.paymentMethod,
          payments: payments,
          discount: Number(discount) || 0,
          taxAmount: payload.taxAmount,
          finalAmount: finalTotal,
          staffId: payload.staffId,
          invoiceDate: moment().toISOString(),
          items: items,
        });
        responseData = res.data?.data;
      } else {
        const res = await api.post("/appointments", payload);
        responseData = res.data?.data;
      }

      const invoiceNo =
        responseData?.invoice?.invoiceNumber ||
        responseData?.invoiceNumber ||
        responseData?.appointmentNumber ||
        "N/A";

      const shopNameVal = selectedShopId
        ? branches.find((b) => b._id === selectedShopId)?.name
        : branches[0]?.name || user?.salonName || "Salon";
      const shopLocVal = selectedShopId
        ? branches.find((b) => b._id === selectedShopId)?.address
        : user?.shop?.address || "";
      const shopPhoneVal = selectedShopId
        ? branches.find((b) => b._id === selectedShopId)?.phone
        : user?.phone || "";
      const sName =
        selectedStaffId === "admin"
          ? user?.name || "Owner"
          : staff.find((s) => s._id === selectedStaffId)?.name || "";

      setSuccessData({
        amount: responseData?.finalAmount || finalTotal,
        invoiceNo,
        paymentMethod: payload.paymentMethod,
        paymentTime: new Date(),
        receiptPayload: {
          appointmentId: responseData?.appointment?._id || bookingId,
          userId: selectedCustomer?._id,
          staffId: selectedStaffId,
          items: cart.map((item) => ({
            product: item.itemType === "product" ? item._id : null,
            serviceId: item.itemType === "service" ? item._id : null,
            variantId:
              item.variantId && item.variantId !== "base"
                ? item.variantId
                : null,
            productName: item.name,
            quantity: item.qty,
            price: item.price,
            amount: item.price * item.qty,
            taxPercent:
              item.itemType === "service"
                ? item.taxPercentage
                : item.taxPercent,
            itemType: item.itemType,
            variantName: item.variantName || null,
          })),
          paymentMethod: payload.paymentMethod,
          payments: payments,
          subTotal: cartCalc.subtotal,
          finalTotal: finalTotal,
          taxAmount: cartCalc.totalTax,
          discount: Number(discount) || 0,
          customerName:
            selectedCustomer?.name ||
            (newCustData.name !== "LOCAL CUSTOMER" ? newCustData.name : ""),
          shopInfo: {
            name: shopNameVal,
            address: shopLocVal,
            phone: shopPhoneVal,
            gstin: user?.shop?.gstin || "",
          },
          staffName: sName,
        },
      });
      setShowSuccessPopup(true);

      // Refetch products to update stock (including variant stock) instantly
      try {
        const prodRes = await api.get("/products");
        if (prodRes.data?.success) {
          setProducts(prodRes.data.data);
          if (metadataCache) metadataCache.products = prodRes.data.data;
        }
      } catch (stockErr) {
        console.error("Could not refresh product stock.", stockErr);
      }

      setCart([]);
      setSelectedCustomer(null);
      setDiscount("");
      setPayments([{ method: paymentMethods[0]?.name || "CASH", amount: "" }]);
      navigate(location.pathname, { replace: true });
    } catch (err) {
      showSnack(err.response?.data?.message || "Checkout Failed", "error");
    } finally {
      setLoading(false);
    }
  };

  try {
    return (
      <div className="h-[calc(100vh-64px)] flex flex-col md:flex-row gap-0 relative">
        {/* Snackbar */}
        <Snackbar
          message={snack.message}
          type={snack.type}
          onClose={() => setSnack({ message: "", type: "success" })}
        />
        <PaymentSuccessPopup
          isOpen={showSuccessPopup}
          onClose={() => setShowSuccessPopup(false)}
          amount={successData?.amount}
          invoiceNo={successData?.invoiceNo}
          paymentMethod={successData?.paymentMethod}
          paymentTime={successData?.paymentTime}
          currencySymbol={currencySymbol}
          receiptPayload={successData?.receiptPayload}
          onShare={() => {
            if (navigator.share) {
              navigator
                .share({
                  title: "Invoice",
                  text: `Invoice ${successData?.invoiceNo} for ${currencySymbol}${successData?.amount}`,
                })
                .catch(console.error);
            } else {
              showSnack("Share not supported on this device", "warning");
            }
          }}
        />
        {/* LEFT COLUMN: Service/Product Selection */}
        <div
          className={`flex-1 flex flex-col min-w-0 bg-transparent h-full ${showMobileCart ? "hidden md:flex" : "flex"}`}
        >
          {/* Top Header */}
          <div className="bg-white dark:bg-dark-surface p-3 border-b border-gray-100 dark:border-dark-border shrink-0 space-y-2">
            {/* Mobile: Departments on top */}
            {activeTab === "services" && departments.length > 0 && (
              <div className="flex items-center gap-1 overflow-x-auto scrollbar-hide md:hidden">
                {departments.map((dept) => (
                  <button
                    key={dept._id}
                    onClick={() => setActiveDept(dept.name)}
                    className={`px-3 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeDept === dept.name ? "bg-brand-primary text-white" : "bg-gray-100 dark:bg-dark-card text-gray-400 hover:bg-gray-200 dark:hover:bg-dark-border hover:text-gray-600"}`}
                  >
                    {dept.name}
                  </button>
                ))}
              </div>
            )}

            {/* Search + Fav + Desktop Departments */}
            <div className="flex items-center gap-3">
              <div className="relative flex-1">
                <SearchIcon
                  className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400"
                  size={14}
                />
                <input
                  type="text"
                  placeholder={
                    activeTab === "services"
                      ? "Search services..."
                      : "Search products..."
                  }
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  className="w-full bg-gray-50/50 dark:bg-dark-card/50 rounded-xl pl-9 pr-4 py-2 text-[12px] font-bold text-gray-700 dark:text-gray-300 border border-transparent focus:border-brand-primary/20 focus:bg-white dark:focus:bg-dark-surface outline-none transition-all placeholder:text-gray-300 dark:placeholder:text-gray-500"
                />
              </div>
              {activeTab === "services" && (
                <button
                  onClick={() => setShowOnlyFavourites(!showOnlyFavourites)}
                  className={`p-2 rounded-xl transition-all border ${showOnlyFavourites ? "bg-red-50 text-red-500 border-red-100 shadow-sm" : "bg-gray-50 text-gray-300 border-transparent hover:bg-gray-100"}`}
                  title={
                    showOnlyFavourites
                      ? "Show all services"
                      : "Show favourite services"
                  }
                >
                  <Heart
                    size={16}
                    fill={showOnlyFavourites ? "currentColor" : "none"}
                  />
                </button>
              )}

              {activeTab === "services" && shopSettings.showServices && (
                <button
                  onClick={() => {
                    setBookingData({
                      date: moment().format("YYYY-MM-DD"),
                      time: "",
                    });
                    setIsBookingModalOpen(true);
                  }}
                  className="p-2 rounded-xl transition-all border bg-gray-50 text-gray-300 border-transparent hover:bg-gray-100 dark:bg-dark-card hover:text-brand-primary"
                  title="Book Appointment"
                >
                  <Calendar size={16} />
                </button>
              )}

              {/* Desktop only: Departments inline */}
              {activeTab === "services" && departments.length > 0 && (
                <div className="hidden md:flex items-center gap-1">
                  <div className="w-px h-7 bg-gray-200 dark:bg-dark-border mx-2 shrink-0" />
                  {departments.map((dept) => (
                    <button
                      key={dept._id}
                      onClick={() => setActiveDept(dept.name)}
                      className={`px-3 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeDept === dept.name ? "bg-brand-primary text-white" : "bg-gray-100 dark:bg-dark-card text-gray-400 hover:bg-gray-200 dark:hover:bg-dark-border hover:text-gray-600"}`}
                    >
                      {dept.name}
                    </button>
                  ))}
                </div>
              )}
            </div>
          </div>

          {/* Grid Scrollable Area */}
          <div className="flex-1 overflow-y-auto px-3 py-2 scrollbar-hide pb-36 md:pb-32">
            {activeTab === "services" ? (
              loading ? (
                <ServicesSkeleton count={12} />
              ) : filteredServices.length > 0 ? (
                <div className="grid grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4 pb-4">
                  {filteredServices.map((service) => (
                    <ServiceCard
                      key={service._id}
                      service={service}
                      displayPrice={getServicePrice(service)}
                      onAdd={addServiceToCart}
                      currencySymbol={currencySymbol}
                    />
                  ))}
                </div>
              ) : (
                <div className="h-full flex flex-col items-center justify-center text-gray-400 opacity-60">
                  <SearchIcon size={48} className="mb-4" />
                  <p>No services found</p>
                </div>
              )
            ) : loading ? (
              <ServicesSkeleton count={12} />
            ) : filteredProducts.length > 0 ? (
              <div className="grid grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4 pb-4">
                {filteredProducts.map((product) => {
                  return (
                    <ProductCard
                      key={product._id}
                      product={product}
                      cart={cart}
                      onAdd={addProductToCart}
                      currencySymbol={currencySymbol}
                      selectedShopId={selectedShopId}
                    />
                  );
                })}
              </div>
            ) : (
              <div className="h-full flex flex-col items-center justify-center text-gray-400 opacity-50">
                <Package size={48} className="mb-4" />
                <p>No products found in inventory</p>
              </div>
            )}
          </div>

          {/* Desktop: Bottom Filter Card - Single Row */}
          <div className="hidden md:block sticky bottom-0 left-0 right-0 z-30 bg-white dark:bg-dark-surface border-t border-gray-100 dark:border-dark-border shadow-[0_-4px_20px_rgba(0,0,0,0.06)] px-3 py-2.5">
            <div className="flex items-center gap-0 overflow-x-auto scrollbar-hide">
              {/* Services / Products Toggle */}
              {shopSettings.showServices && shopSettings.showProducts && (
                <div className="flex items-center gap-1 shrink-0">
                  <button
                    onClick={() => {
                      setActiveTab("services");
                      setSearchQuery("");
                    }}
                    className={`flex items-center gap-1 px-3 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeTab === "services" ? "bg-brand-primary text-white shadow-sm" : "bg-gray-100 dark:bg-dark-card text-gray-500 hover:text-gray-700"}`}
                  >
                    <FileText size={12} /> Services
                  </button>
                  <button
                    onClick={() => {
                      setActiveTab("products");
                      setSearchQuery("");
                    }}
                    className={`flex items-center gap-1 px-3 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeTab === "products" ? "bg-brand-primary text-white shadow-sm" : "bg-gray-100 text-gray-500 hover:text-gray-700"}`}
                  >
                    <Package size={12} /> Products
                  </button>
                </div>
              )}

              {/* Vertical Divider - Only show if both are enabled and toggle is visible */}
              {shopSettings.showServices && shopSettings.showProducts && (
                <div className="w-px h-7 bg-gray-200 dark:bg-dark-border mx-2.5 shrink-0" />
              )}

              {/* Categories */}
              <div className="flex items-center gap-1 overflow-x-auto scrollbar-hide">
                {activeTab === "services" ? (
                  <>
                    <button
                      onClick={() => setActiveSvcCat("All")}
                      className={`px-2.5 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeSvcCat === "All" ? "bg-brand-primary/10 text-brand-primary" : "bg-gray-100 text-gray-400 hover:text-gray-600"}`}
                    >
                      All
                    </button>
                    {serviceCategories.map((cat) => (
                      <button
                        key={cat._id}
                        onClick={() => setActiveSvcCat(cat.name)}
                        className={`px-2.5 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeSvcCat === cat.name ? "bg-brand-primary/10 text-brand-primary" : "bg-gray-100 text-gray-400 hover:text-gray-600"}`}
                      >
                        {cat.name}
                      </button>
                    ))}
                  </>
                ) : (
                  <>
                    <button
                      onClick={() => setActiveProdCat("All")}
                      className={`px-2.5 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeProdCat === "All" ? "bg-brand-primary/10 text-brand-primary" : "bg-gray-100 text-gray-400 hover:text-gray-600"}`}
                    >
                      All
                    </button>
                    {productCategories.map((cat) => (
                      <button
                        key={cat._id}
                        onClick={() => setActiveProdCat(cat.name)}
                        className={`px-2.5 py-1.5 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all whitespace-nowrap ${activeProdCat === cat.name ? "bg-brand-primary/10 text-brand-primary" : "bg-gray-100 text-gray-400 hover:text-gray-600"}`}
                      >
                        {cat.name}
                      </button>
                    ))}
                  </>
                )}
              </div>
            </div>
          </div>

          {/* Mobile: Filter FAB */}
          <button
            onClick={() => setShowFilterPopup(true)}
            className={`md:hidden fixed z-[41] w-12 h-12 rounded-full bg-brand-primary text-white shadow-lg shadow-brand-primary/30 flex items-center justify-center active:scale-95 transition-all ${showMobileCart ? "hidden" : ""} ${cart.length > 0 ? "bottom-[10rem]" : "bottom-20"} right-4`}
          >
            <SlidersHorizontal size={18} />
          </button>

          {/* Mobile: Filter Overlay Popup - Bottom Right Card */}
          {showFilterPopup && (
            <>
              <div
                className="md:hidden fixed inset-0 z-[45] bg-black/30"
                onClick={() => setShowFilterPopup(false)}
              />
              <div
                className={`md:hidden fixed z-[46] right-3 ${cart.length > 0 ? "bottom-[13.5rem]" : "bottom-[5.5rem]"} w-56 bg-white dark:bg-dark-surface rounded-xl shadow-2xl border border-gray-100 dark:border-dark-border p-3 space-y-3`}
              >
                {/* Services / Products */}
                {shopSettings.showServices && shopSettings.showProducts && (
                  <div className="space-y-1.5">
                    <span className="text-[9px] font-bold text-gray-400 uppercase tracking-widest">
                      Type
                    </span>
                    <button
                      onClick={() => {
                        setActiveTab("services");
                        setSearchQuery("");
                      }}
                      className={`w-full flex items-center gap-2 px-3 py-2 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all ${activeTab === "services" ? "bg-brand-primary text-white" : "bg-gray-50 text-gray-500"}`}
                    >
                      <FileText size={13} /> Services
                    </button>
                    <button
                      onClick={() => {
                        setActiveTab("products");
                        setSearchQuery("");
                      }}
                      className={`w-full flex items-center gap-2 px-3 py-2 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all ${activeTab === "products" ? "bg-brand-primary text-white" : "bg-gray-50 text-gray-500"}`}
                    >
                      <Package size={13} /> Products
                    </button>
                  </div>

                {shopSettings.showServices && shopSettings.showProducts && (
                  <div className="h-px bg-gray-100 dark:bg-dark-border" />
                )}

                {/* Categories */}
                <div className="space-y-1.5">
                  <span className="text-[9px] font-bold text-gray-400 uppercase tracking-widest">
                    Category
                  </span>
                  <div className="flex flex-col gap-1 max-h-48 overflow-y-auto scrollbar-hide">

                        <button
                          onClick={() => {

                            setShowFilterPopup(false);

                        >
                          All
                        </button>
                        {serviceCategories.map((cat) => (
                          <button
                            key={cat._id}

                              setActiveSvcCat(cat.name);
                              setShowFilterPopup(false);
                            }}
                            className={`w-full text-left px-3 py-2 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all ${activeSvcCat === cat.name ? "bg-brand-primary/10 text-brand-primary" : "bg-gray-50 text-gray-400"}`}
                          >

                            key={cat._id}

                              setActiveProdCat(cat.name);
                              setShowFilterPopup(false);
                            }}
                            className={`w-full text-left px-3 py-2 rounded-lg text-[10px] font-black uppercase tracking-wider transition-all ${activeProdCat === cat.name ? "bg-brand-primary/10 text-brand-primary" : "bg-gray-50 text-gray-400"}`}

              <button

                View Cart

        <div

        >

            <button

              className="flex items-center gap-2 text-gray-600 font-bold text-[11px] uppercase tracking-wider hover:text-brand-primary transition-colors"
            >
              <ArrowLeft size={16} /> Back to Items
            </button>
          </div>

          {/* Header & Selectors */}
          <div className="p-4 bg-white dark:bg-dark-surface border-b border-gray-100 dark:border-dark-border flex flex-col gap-3">
            {(isAdmin ||
              (user?.role === "staff" ? false : branches.length > 1)) && (
              <div className="flex gap-2">
                {(branches.length > 1 || isAdmin || user?.role === "staff") && (
                  <div className="flex-1 min-w-0">
                    <CustomDropdown
                      variant="compact"
                      icon={Store}
                      value={selectedShopId}
                      onChange={(e) => setSelectedShopId(e.target.value)}
                      options={branches
                        .filter((shop) => {
                          if (user?.role === "staff") {
                            const userBranchId =
                              typeof user.branchId === "string"
                                ? user.branchId
                                : user.branchId?._id;
                            return shop._id === userBranchId;
                          }
                          return true;
                        })
                        .map((shop) => ({
                          value: shop._id,
                          label: `${shop.name}${shop.address ? ` (${shop.address})` : ""}`,
                        }))}
                      placeholder="Select Branch"
                    />
                  </div>
                )}
                {isAdmin && (
                  <div className="flex-1 min-w-0">
                    <CustomDropdown
                      variant="compact"
                      icon={Users}
                      value={selectedStaffId}
                      onChange={(e) => {
                        setSelectedStaffId(e.target.value);
                      }}
                      options={[
                        {
                          value: "admin",
                          label: user?.name || "Owner",
                        },
                        ...filteredStaff.map((s) => ({
                          value: s._id,
                          label: s.name,
                        })),
                      ]}
                      placeholder="Select Staff"
                    />
                  </div>
                )}
              </div>
            )}

            {/* Customer Selection */}
            <div className="flex items-center gap-3">
              <div className="flex-1 relative">
                {selectedCustomer ? (
                  <div className="bg-brand-primary/5 p-2 rounded-xl flex items-center justify-between border border-brand-primary/10">
                    <div className="flex items-center gap-2 overflow-hidden">
                      <div className="w-8 h-8 rounded-full bg-white flex items-center justify-center text-brand-primary font-black text-xs shrink-0">
                        {selectedCustomer.name?.charAt(0) || "?"}
                      </div>
                      <div className="leading-tight truncate">
                        <p className="font-bold text-[11px] text-gray-800 dark:text-gray-200 truncate">
                          {selectedCustomer.name || "Unnamed Customer"}
                        </p>
                        <p className="text-[9px] text-gray-400">
                          {selectedCustomer.phone}
                          {(selectedCustomer.walletBalance || 0) !== 0 && (
                            <span
                              className={`ml-2 font-bold whitespace-nowrap ${selectedCustomer.walletBalance > 0 ? "text-emerald-500" : "text-rose-500"}`}
                            >
                              {selectedCustomer.walletBalance > 0
                                ? "Wallet:"
                                : "Due:"}{" "}
                              <span className="notranslate">
                                {currencySymbol}
                                {Math.abs(
                                  selectedCustomer.walletBalance,
                                ).toLocaleString()}
                              </span>
                            </span>
                          )}
                        </p>
                      </div>
                    </div>
                    <button
                      onClick={() => setSelectedCustomer(null)}
                      className="p-1 hover:text-red-500 text-gray-300"
                    >
                      <X size={14} />
                    </button>
                  </div>
                ) : (
                  <div className="relative">
                    <CustomTextField
                      variant="compact"
                      placeholder="Find Customer..."
                      icon={SearchIcon}
                      value={customerSearch}
                      onChange={(e) => {
                        setCustomerSearch(e.target.value);
                        setShowCustomerResults(true);
                      }}
                      onFocus={() => setShowCustomerResults(true)}
                    />
                    {showCustomerResults && customerSearch && (
                      <div className="absolute top-full left-0 right-0 mt-1 bg-white dark:bg-dark-surface rounded-xl shadow-xl border border-gray-100 dark:border-dark-border z-50 max-h-48 overflow-y-auto">
                        {filteredCustomers.length > 0 ? (
                          filteredCustomers.map((c) => (
                            <button
                              key={c._id}
                              onClick={() => handleCustomerSelect(c)}
                              className="w-full text-left px-3 py-2 hover:bg-gray-50 dark:hover:bg-dark-card border-b border-gray-50 dark:border-dark-border last:border-none"
                            >
                              <p className="text-[11px] font-bold text-gray-700 dark:text-gray-300">
                                {c.name}
                              </p>
                              <p className="text-[9px] text-gray-400">
                                {c.phone}
                              </p>
                            </button>
                          ))
                        ) : (
                          <div className="p-3 text-center text-gray-400 text-[10px]">
                            <button
                              onClick={() => setIsCustModalOpen(true)}
                              className="text-brand-primary font-bold"
                            >
                              Create "{customerSearch}"
                            </button>
                          </div>
                        )}
                      </div>
                    )}
                  </div>
                )}
              </div>
              {!selectedCustomer && (
                <button
                  onClick={() => setIsCustModalOpen(true)}
                  className="p-2 bg-gray-50 dark:bg-dark-card rounded-xl text-brand-primary hover:bg-gray-100 dark:hover:bg-dark-border transition-colors"
                >
                  <Plus size={16} />
                </button>
              )}
            </div>
          </div>

          {/* Cart Items */}
          <div className="flex-1 overflow-y-auto p-4 scrollbar-hide bg-white dark:bg-dark-surface">
            {cart.length > 0 ? (
              <div className="space-y-2">
                {cart.map((item) => {
                  const lineTotal = item.price * item.qty;
                  let lineTax = 0,
                    lineDisplay = lineTotal;
                  if (item.itemType === "service") {
                    const taxPct = item.taxPercentage || 0;
                    if (taxPct > 0) {
                      lineTax = item.isTaxInclusive
                        ? lineTotal - lineTotal / (1 + taxPct / 100)
                        : lineTotal * (taxPct / 100);
                    }
                    lineDisplay = item.isTaxInclusive
                      ? lineTotal
                      : lineTotal + lineTax;
                  } else {
                    const taxPct = item.taxPercent || 0;
                    if (taxPct > 0) {
                      lineTax = item.isTaxInclusive
                        ? lineTotal - lineTotal / (1 + taxPct / 100)
                        : lineTotal * (taxPct / 100);
                    }
                    lineDisplay = item.isTaxInclusive
                      ? lineTotal
                      : lineTotal + lineTax;
                  }

                  return (
                    <div
                      key={item.variantKey}
                      className="p-2 bg-gray-50/50 dark:bg-dark-card/50 rounded-xl border border-gray-100 dark:border-dark-border flex items-center gap-2 transition-all"
                    >
                      <div className="flex-1 min-w-0">
                        <h4 className="font-bold text-[10px] text-gray-700 dark:text-gray-300 truncate">
                          {item.name}
                          {item.variantName ? ` - ${item.variantName}` : ""}
                        </h4>
                        {item.itemType === "product" && (
                          <div className="text-[8px] text-gray-400 font-bold mb-0.5 mt-0.5">
                            Available: {item.currentStock - item.qty}
                          </div>
                        )}
                        <div className="flex items-center gap-1.5">
                          <span
                            className={`text-[8px] font-black px-1.5 py-0.5 rounded uppercase tracking-widest ${item.itemType === "service" ? "bg-blue-50 text-blue-500 border border-blue-100" : "bg-purple-50 text-purple-500 border border-purple-100"}`}
                          >
                            {item.itemType === "service"
                              ? item.deptName || "Service"
                              : "Product"}
                            {item.itemType === "service" &&
                              item.isDeptMatched && (
                                <span className="ml-1 text-[7px] text-emerald-500">
                                  Γ£ô
                                </span>
                              )}
                          </span>
                          {((item.itemType === "service" &&
                            (item.taxPercentage || 0) > 0) ||
                            (item.itemType === "product" &&
                              (item.taxPercent || item.taxPercentage || 0) >
                                0)) && (
                            <span className="text-[8px] font-bold px-1.5 py-0.5 rounded bg-emerald-50 text-emerald-600 border border-emerald-100">
                              {item.itemType === "service"
                                ? `${item.taxName || "Tax"} ${item.taxPercentage}%`
                                : `${item.taxName || "Tax"} ${item.taxPercent || item.taxPercentage}%`}
                            </span>
                          )}
                        </div>
                      </div>
                      <div className="flex items-center gap-1.5">
                        <div className="flex items-center gap-2 bg-white dark:bg-dark-surface rounded-xl p-1 border border-gray-100 dark:border-dark-border shadow-sm">
                          <button
                            onClick={() => updateQty(item.variantKey, -1)}
                            className="w-7 h-7 flex items-center justify-center text-gray-400 hover:text-red-500 hover:bg-red-50 rounded-lg transition-colors"
                          >
                            <Minus size={12} />
                          </button>
                          <span className="text-[12px] font-black text-gray-800 dark:text-gray-200 min-w-[1.2rem] text-center">
                            {item.qty}
                          </span>
                          <button
                            onClick={() => updateQty(item.variantKey, 1)}
                            className="w-7 h-7 flex items-center justify-center text-gray-400 hover:text-brand-primary hover:bg-brand-primary/5 rounded-lg transition-colors"
                          >
                            <Plus size={12} />
                          </button>
                        </div>
                        <div className="text-right min-w-[40px]">
                          <p className="text-gray-800 dark:text-gray-200 font-black text-[10px]">
                            <span className="notranslate">
                              {currencySymbol}
                              {Math.round(lineDisplay).toLocaleString()}
                            </span>
                          </p>
                          {lineTax > 0 && (
                            <p
                              className={`text-[8px] font-bold ${item.isTaxInclusive ? "text-gray-400" : "text-emerald-500"}`}
                            >
                              {item.isTaxInclusive ? "Incl. Tax: " : "Tax: +"}
                              <span className="notranslate">
                                {currencySymbol}
                                {Math.round(lineTax).toLocaleString()}
                              </span>
                            </p>
                          )}
                        </div>
                        <button
                          onClick={() => removeFromCart(item.variantKey)}
                          className="text-gray-300 hover:text-red-500 transition-colors"
                        >
                          <X size={14} />
                        </button>
                      </div>
                    </div>
                  );
                })}
              </div>
            ) : (
              <div className="h-full flex flex-col items-center justify-center text-gray-300 space-y-2 opacity-50">
                <ShoppingCart size={24} />
                <p className="text-[10px] font-bold uppercase tracking-widest">
                  Empty Cart
                </p>
              </div>
            )}
          </div>

          {/* Footer / Payment */}
          <div className="p-4 bg-white dark:bg-dark-surface border-t border-gray-50 dark:border-dark-border mt-auto space-y-3">
            <div className="space-y-2">
              <div className="flex items-center justify-between px-1">
                <span className="text-[10px] font-black text-gray-400 uppercase tracking-[0.15em]">
                  Payment
                </span>
                <button
                  onClick={handleAddPaymentRow}
                  className="text-[10px] font-black text-brand-primary hover:text-brand-secondary flex items-center gap-1.5 transition-colors"
                >
                  <Plus size={12} />
                  <span>ADD SPLIT</span>
                </button>
              </div>
              <div className="overflow-visible pr-1 space-y-2">
                {payments.map((p, idx) => (
                  <div key={idx} className="flex gap-2 items-center">
                    <div className="flex-1">
                      <CustomDropdown
                        variant="compact"
                        value={p.method}
                        onChange={(e) =>
                          handleUpdatePayment(idx, "method", e.target.value)
                        }
                        options={paymentMethods.map((pm) => ({
                          value: pm.name,
                          label: pm.name,
                        }))}
                        placeholder="Method"
                      />
                    </div>
                    <div className="w-24 bg-gray-50/50 dark:bg-dark-card/50 rounded-xl px-3 py-2 flex items-center">
                      <span className="text-[10px] text-gray-400 font-bold mr-1 notranslate">
                        {currencySymbol}
                      </span>
                      <input
                        type="number"
                        value={p.amount}
                        onChange={(e) =>
                          handleUpdatePayment(idx, "amount", e.target.value)
                        }
                        className="w-full bg-transparent text-[11px] font-black text-gray-700 dark:text-gray-300 outline-none text-right"
                        placeholder="0"
                      />
                    </div>
                    {payments.length > 1 && (
                      <button
                        onClick={() => handleRemovePaymentRow(idx)}
                        className="text-gray-300 hover:text-red-500 p-1"
                      >
                        <Trash2 size={14} />
                      </button>
                    )}
                  </div>
                ))}
              </div>
            </div>

            {/* Totals */}
            <div className="bg-gray-50/50 dark:bg-dark-card/50 p-3 rounded-xl border border-gray-100 dark:border-dark-border">
              <div className="flex items-center justify-between text-[10px] mb-1">
                <span className="font-bold text-gray-500 dark:text-gray-400 uppercase">
                  Subtotal
                </span>
                <span className="font-black text-gray-700 dark:text-gray-300">
                  <span className="notranslate">
                    {currencySymbol}
                    {Math.round(cartCalc.subtotal).toLocaleString()}
                  </span>
                </span>
              </div>
              {cartCalc.totalTax > 0 && (
                <div className="flex items-center justify-between text-[10px] mb-1">
                  <span className="font-bold text-emerald-600 uppercase">
                    Tax
                  </span>
                  <span className="font-black text-emerald-600">
                    <span className="notranslate">
                      {cartCalc.hasInclusiveTax ? "" : "+"}
                      {currencySymbol}
                      {Math.round(cartCalc.totalTax).toLocaleString()}
                    </span>
                  </span>
                </div>
              )}
              <div className="flex items-center justify-between text-[10px] mb-2">
                <span className="font-bold text-gray-500 dark:text-gray-400 uppercase">
                  Discount
                </span>
                <div className="flex items-center w-16 bg-white dark:bg-dark-surface rounded border border-gray-200 dark:border-dark-border px-1 py-0.5">
                  <span className="text-[9px] text-gray-400 mr-1 notranslate">
                    {currencySymbol}
                  </span>
                  <input
                    type="number"
                    value={discount}
                    onChange={(e) => setDiscount(e.target.value)}
                    className="w-full bg-transparent outline-none font-bold text-right text-[10px]"
                  />
                </div>
              </div>

              {/* Wallet Usage */}
              {selectedCustomer &&
                (selectedCustomer.walletBalance || 0) > 0 &&
                finalTotal > 0 &&
                payments.every((p) => p.method !== "WALLET") && (
                  <div className="mb-4 bg-brand-primary/5 border border-brand-primary/20 rounded-lg p-3 flex flex-col gap-2">
                    <div className="flex items-center justify-between">
                      <span className="text-[10px] font-black text-brand-primary uppercase tracking-widest flex items-center gap-1.5">
                        <Wallet size={12} /> Wallet Balance
                      </span>
                      <span className="text-[11px] font-black text-brand-primary notranslate">
                        {currencySymbol}
                        {(selectedCustomer.walletBalance || 0).toLocaleString()}
                      </span>
                    </div>
                    <button
                      onClick={() => {
                        const useAmt = Math.min(
                          finalTotal,
                          selectedCustomer.walletBalance || 0,
                        );
                        if (useAmt >= finalTotal) {
                          setPayments([{ method: "WALLET", amount: useAmt }]);
                        } else {
                          const ewi = payments.findIndex(
                            (p) => p.method === "WALLET",
                          );
                          if (ewi >= 0) {
                            handleUpdatePayment(ewi, "amount", useAmt);
                          } else if (
                            payments.length === 1 &&
                            (payments[0].amount === "" ||
                              Number(payments[0].amount) === 0)
                          ) {
                            setPayments([{ method: "WALLET", amount: useAmt }]);
                          } else {
                            setPayments([
                              ...payments,
                              { method: "WALLET", amount: useAmt },
                            ]);
                          }
                        }
                      }}
                      className="w-full py-1.5 bg-brand-primary hover:bg-brand-secondary text-white text-[10px] font-bold uppercase tracking-widest rounded-md transition-colors shadow-sm"
                    >
                      Use Wallet to Pay
                    </button>
                  </div>
                )}

              <div className="flex items-center justify-between border-t border-gray-200 dark:border-dark-border pt-2">
                <span className="font-black text-gray-800 dark:text-gray-200 uppercase text-xs">
                  Total Payable
                </span>
                <span className="font-black text-brand-primary text-lg notranslate">
                  {currencySymbol}
                  {finalTotal.toLocaleString()}
                </span>
              </div>

              {remainingAmount > 0 && (
                <div className="mt-3 bg-amber-50 border border-amber-100 rounded-lg px-3 py-2 flex items-center justify-between">
                  <span className="text-[10px] font-black text-amber-600 uppercase tracking-widest">
                    Unpaid:{" "}
                    <span className="notranslate">
                      {currencySymbol}
                      {remainingAmount.toLocaleString()}
                    </span>
                  </span>
                  {!selectedCustomer && (
                    <span className="text-[8px] font-bold text-red-500 uppercase tracking-wider">
                      *Select Customer
                    </span>
                  )}
                </div>
              )}
              {remainingAmount < 0 && (
                <div className="mt-3 bg-emerald-50 border border-emerald-100 rounded-lg px-3 py-2 flex items-center justify-between">
                  <span className="text-[10px] font-black text-emerald-600 uppercase tracking-widest">
                    Overpaid:{" "}
                    <span className="notranslate">
                      {currencySymbol}
                      {Math.abs(remainingAmount).toLocaleString()}
                    </span>
                  </span>
                  {!selectedCustomer && (
                    <span className="text-[8px] font-bold text-red-500 uppercase tracking-wider">
                      *Select Customer
                    </span>
                  )}
                </div>
              )}
            </div>

            <button
              onClick={handleCheckout}
              disabled={
                loading ||
                cart.length === 0 ||
                (remainingAmount !== 0 && !selectedCustomer)
              }
              className={`w-full font-black py-2.5 rounded-xl transition-all disabled:opacity-50 disabled:cursor-not-allowed shadow-lg flex items-center justify-center gap-2 uppercase tracking-widest text-xs mt-3 ${
                loading || cart.length === 0
                  ? "bg-gray-200 text-gray-400 cursor-not-allowed"
                  : remainingAmount > 0
                    ? "bg-amber-500 hover:bg-amber-600 text-white shadow-amber-500/20 active:scale-[0.98]"
                    : remainingAmount < 0
                      ? "bg-emerald-500 hover:bg-emerald-600 text-white shadow-emerald-500/20 active:scale-[0.98]"
                      : "bg-brand-primary text-white hover:bg-brand-secondary active:scale-[0.98] shadow-brand-primary/20"
              }`}
            >
              {loading ? (
                <Clock className="animate-spin" size={16} />
              ) : remainingAmount < 0 ? (
                <Wallet size={16} />
              ) : (
                <CheckCircle2 size={16} />
              )}
              {loading
                ? "Processing..."
                : remainingAmount > 0
                  ? "DUE & SAVE INVOICE"
                  : remainingAmount < 0
                    ? "WALLET & SAVE INVOICE"
                    : "SAVE INVOICE"}
            </button>
          </div>
        </div>

        {/* Create Customer Modal */}
        <Modal
          isOpen={isCustModalOpen}
          onClose={() => setIsCustModalOpen(false)}
          title="Add New Customer"
          maxWidth="max-w-md"
          footer={
            <div className="flex gap-3">
              <button
                onClick={() => setIsCustModalOpen(false)}
                className="flex-1 py-2.5 bg-gray-100 dark:bg-dark-card text-gray-700 dark:text-gray-300 rounded-xl hover:bg-gray-200 dark:hover:bg-dark-border font-bold text-sm transition-colors"
              >
                Cancel
              </button>
              <button
                onClick={handleCreateCustomer}
                className="flex-1 py-2.5 bg-brand-primary text-white rounded-xl hover:bg-brand-secondary font-bold text-sm shadow-lg shadow-brand-primary/20 transition-colors"
              >
                Create Customer
              </button>
            </div>
          }
        >
          <form onSubmit={handleCreateCustomer} className="space-y-4 py-2">
            <CustomTextField
              label="Full Name"
              placeholder="e.g. Rahul Sharma"
              icon={User}
              value={newCustData.name}
              onChange={(e) =>
                setNewCustData({ ...newCustData, name: e.target.value })
              }
              required
            />
            <CustomTextField
              label="Phone Number"
              placeholder="9876543210"
              icon={Phone}
              value={newCustData.phone}
              onChange={(e) =>
                setNewCustData({ ...newCustData, phone: e.target.value })
              }
              required
            />
            <CustomTextField
              label="Email (Optional)"
              placeholder="client@example.com"
              icon={Mail}
              value={newCustData.email}
              onChange={(e) =>
                setNewCustData({ ...newCustData, email: e.target.value })
              }
            />
          </form>
        </Modal>

        {/* Booking Modal */}
        <Modal
          isOpen={isBookingModalOpen}
          onClose={() => setIsBookingModalOpen(false)}
          title="Book Appointment"
          maxWidth="max-w-md"
          footer={
            <div className="flex gap-3">
              <button
                onClick={() => setIsBookingModalOpen(false)}
                className="flex-1 py-2.5 bg-gray-100 dark:bg-dark-card text-gray-700 dark:text-gray-300 rounded-xl hover:bg-gray-200 dark:hover:bg-dark-border font-bold text-sm transition-colors uppercase tracking-widest"
              >
                Cancel
              </button>
              <button
                onClick={handleBookAppointment}
                disabled={
                  loading ||
                  cart.filter((i) => i.itemType === "service").length === 0
                }
                className="flex-1 py-2.5 bg-brand-primary text-white rounded-xl hover:bg-brand-secondary font-bold text-sm shadow-lg shadow-brand-primary/20 transition-colors uppercase tracking-widest disabled:opacity-50"
              >
                {loading ? "Booking..." : "Book Appointment"}
              </button>
            </div>
          }
        >
          <div className="space-y-4 py-2">
            {/* Customer Info */}
            <div className="bg-gray-50 dark:bg-dark-card p-3 rounded-2xl border border-gray-100 dark:border-dark-border">
              <p className="text-[10px] font-black text-gray-400 uppercase tracking-widest mb-1">
                Customer
              </p>
              {selectedCustomer ? (
                <div className="flex items-center justify-between">
                  <div>
                    <p className="text-sm font-bold text-gray-800 dark:text-gray-200">
                      {selectedCustomer.name}
                    </p>
                    <p className="text-xs text-gray-500">
                      {selectedCustomer.phone}
                    </p>
                  </div>
                  <button
                    onClick={() => setSelectedCustomer(null)}
                    className="text-red-500 hover:text-red-600"
                  >
                    <Trash2 size={14} />
                  </button>
                </div>
              ) : (
                <p className="text-xs text-amber-600 font-bold italic">
                  No customer selected (LOCAL CUSTOMER)
                </p>
              )}
            </div>

            {/* Services */}
            <div className="bg-gray-50 dark:bg-dark-card p-3 rounded-2xl border border-gray-100 dark:border-dark-border">
              <p className="text-[10px] font-black text-gray-400 uppercase tracking-widest mb-2">
                Services in Cart
              </p>
              <div className="max-h-32 overflow-y-auto space-y-2 pr-1 scrollbar-hide">
                {cart.filter((i) => i.itemType === "service").length > 0 ? (
                  cart
                    .filter((i) => i.itemType === "service")
                    .map((s) => (
                      <div
                        key={s.variantKey}
                        className="flex justify-between items-center text-xs"
                      >
                        <span className="font-bold text-gray-700 dark:text-gray-300">
                          {s.name} x {s.qty}
                        </span>
                        <span className="text-brand-primary font-black">
                          {currencySymbol}
                          {Math.round(s.price * s.qty)}
                        </span>
                      </div>
                    ))
                ) : (
                  <p className="text-[10px] text-red-500 font-bold uppercase text-center py-2">
                    Please add services to cart first
                  </p>
                )}
              </div>
            </div>

            {/* Date & Time */}
            <div className="space-y-4">
              <CustomDatePicker
                label="Appointment Date"
                value={bookingData.date}
                onChange={(val) =>
                  setBookingData({ ...bookingData, date: val })
                }
              />

              <div className="space-y-2">
                <label className="text-[10px] font-black text-gray-400 uppercase tracking-widest ml-1">
                  Select Time Slot {slotLoading && "(Loading...)"}
                </label>
                <div className="grid grid-cols-4 gap-2 max-h-40 overflow-y-auto p-1 scrollbar-hide">
                  {availableSlots.length > 0 ? (
                    availableSlots.map((slot) => {
                      const isSelected = bookingData.time === slot;
                      return (
                        <button
                          key={slot}
                          onClick={() =>
                            setBookingData({ ...bookingData, time: slot })
                          }
                          className={`py-2 px-1 rounded-xl text-[10px] font-black transition-all border ${
                            isSelected
                              ? "bg-brand-primary text-white border-brand-primary shadow-lg shadow-brand-primary/20 scale-[1.02]"
                              : "bg-gray-50 dark:bg-dark-card text-gray-600 dark:text-gray-400 border-gray-100 dark:border-dark-border hover:border-brand-primary/30"
                          }`}
                        >
                          {slot}
                        </button>
                      );
                    })
                  ) : !slotLoading ? (
                    <div className="col-span-4 py-4 text-center bg-red-50 dark:bg-red-900/10 rounded-xl border border-red-100 dark:border-red-900/20">
                      <p className="text-[10px] font-bold text-red-500 uppercase">
                        No slots available for this date
                      </p>
                    </div>
                  ) : (
                    Array(8)
                      .fill(0)
                      .map((_, i) => (
                        <div
                          key={i}
                          className="h-8 bg-gray-100 dark:bg-dark-border rounded-xl animate-pulse"
                        />
                      ))
                  )}
                </div>
              </div>
            </div>

            {/* Branch & Staff */}
            <div className="space-y-4">
              <CustomDropdown
                label="Branch"
                options={branches.map((b) => ({ label: b.name, value: b._id }))}
                value={selectedShopId}
                onChange={(val) => setSelectedShopId(val)}
                icon={Store}
              />
              <CustomDropdown
                label="Assign Staff"
                options={[
                  { label: "Admin (Owner)", value: "admin" },
                  ...filteredStaff.map((s) => ({
                    label: s.name,
                    value: s._id,
                  })),
                ]}
                value={selectedStaffId}
                onChange={setSelectedStaffId}
                icon={Users}
              />
            </div>
          </div>
        </Modal>

        <VariantSelectionModal
          isOpen={showVariantModal}
          onClose={() => setShowVariantModal(false)}
          item={itemForVariants}
          activeDept={activeDept}
          departments={departments}
          currencySymbol={currencySymbol}
          selectedShopId={selectedShopId}
          cart={cart}
          onSelect={(item, price, variantData) => {
            if (item.itemType === "service") {
              addServiceToCart(item, price, variantData);
            } else {
              addProductToCart(item, price, variantData);
            }
          }}
        />
      </div>
    );
  } catch (e) {
    console.error("UnifiedInvoiceEntry Render Error:", e);
    return (
      <div className="p-8 flex flex-col items-center justify-center h-full text-red-500">
        <h2 className="text-xl font-bold mb-2">Something went wrong</h2>
        <p>{e.message}</p>
      </div>
    );
  }
};

export default UnifiedInvoiceEntry;
